lemma left_{r} (N T : ℤ) (s : ℕ) (hn : 14 ≤ 24*N+{r})
    (ht : 0 ≤ 12*T+s) (hs : s < 12) :
    reduced {r} ((s+1)%12) 0 N (T+if s=11 then 1 else 0) 0 ↔
    (¬reduced {r} s {v} N T (2*N+({q})) ∧ (reduced {r} s 0 N T 0 ∨ reduced {r} s 1 N T 0)) ∨
    (reduced {r} s {v} N T (2*N+({q})) ∧ ¬reduced {r} s 0 N T 0) := by
  interval_cases s <;> (simp [reduced, b{r}] <;> ((try split_ifs) <;> omega))
lemma right_{r} (N T : ℤ) (s : ℕ) (hn : 14 ≤ 24*N+{r})
